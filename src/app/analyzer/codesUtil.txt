const STATUS_TIMESTAMP_MAP = {
  aguardando: "aguardando",
  emDoca: "emDoca",
  carregando: "carregando",
  liberado: "liberado",
} as const;

function formatarDuracao(segundos: number): string {
  const h = Math.floor(segundos / 3600);
  const m = Math.floor((segundos % 3600) / 60);
  const s = segundos % 60;

  return [h, m, s]
    .map(v => String(v).padStart(2, "0"))
    .join(":");
}

function obterCorSLA(
  status: StatusCarregamento,
  segundos: number
): string {

  const minutos = segundos / 60;

  if (status === "emDoca") {
    if (minutos <= 15) return "border-green-500";
    if (minutos <= 30) return "border-yellow-500";
    return "border-red-500";
  }

  if (status === "carregando") {
    if (minutos <= 20) return "border-green-500";
    if (minutos <= 30) return "border-yellow-500";
    return "border-red-500";
  }

  return "";
}

        ? new Date(b.timestamp.liberado).getTime()
        : 0;

      return tb - ta;

    const campo = STATUS_TIMESTAMP_MAP[
      status as keyof typeof STATUS_TIMESTAMP_MAP

      : 0;

    const tb = b.timestamp?.[campo]
      ? new Date(b.timestamp[campo]).getTime()
      : 0;

    return ta - tb;
